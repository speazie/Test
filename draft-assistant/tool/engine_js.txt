// Tunable weights. These were previously hardcoded from intuition; they are now
// optimised against simulated championship rate (see optimise.js).
// Forward planning can be turned off, so greedy and planned drafting can be
// run against each other. Both use identical projections, so unlike a
// projection change this comparison is a fair test.
// A function parameter is 'defined' even when nothing was passed for it, so a
// typeof check alone silently turned planning OFF everywhere the harness
// builds an instance -- including every regression test.
var PLAN = (typeof PLAN_OVERRIDE==='undefined'||PLAN_OVERRIDE==null) ? true : !!PLAN_OVERRIDE;
var CFG = (typeof CFG_OVERRIDE!=='undefined') ? CFG_OVERRIDE : {"bye2": 14, "bye3": 26, "bye4": 70, "byeAll": 5, "byeNoCover": 7, "byeCover": 9, "reachFree": 8, "reachRate": 0.6, "cuffMult": 0.45, "cuffTop": 70, "cuff": 40, "conc": 10, "stack": 7, "dropMin": 10, "dropMult": 0.55, "dropCap": 30, "runCap": 22, "runRate": 45, "coldPen": 8, "scarMin": 6, "scarMax": 48, "benchWR": 12, "benchRB": 6, "rb5pen": 14,
  // Previously hardcoded OUTSIDE CFG, so sim/optimise.js never saw them and
  // they were never tested. Defaults below are the exact prior values, so

  "qbEarly": -30, "qbMid": -12, "kRound": 15, "dstRound": 14,
  "flexTE": -8, "survThresh": 0.55, "lateHorizon": 38};
;
const TEAMS=10, ROUNDS=15;
const SLOTS=[["QB",["QB"]],["WR",["WR"]],["WR",["WR"]],["RB",["RB"]],["RB",["RB"]],
             ["TE",["TE"]],["W/R/T",["WR","RB","TE"]],["K",["K"]],["DEF",["DST"]],
             ["BN",null],["BN",null],["BN",null],["BN",null],["BN",null],["BN",null]];

let slot=null, gone=new Set(), mine=[], hist=[], filter="ALL", query="", takenAt={}, goneAt={}, sortMode="value";

const byName=Object.fromEntries(PLAYERS.map(p=>[p.n,p]));
const myPicks=()=>slot?Array.from({length:ROUNDS},(_,i)=>{const r=i+1;
  return r%2?(r-1)*TEAMS+slot:r*TEAMS-slot+1;}):[];
const curPick=()=>gone.size+mine.length+1;
const nextMine=()=>myPicks().find(n=>n>=curPick());
const isMyTurn=()=>slot&&nextMine()===curPick();

// ---- roster shape ----
function fillRoster(){
  const board=SLOTS.map(([c,ok])=>({code:c,ok,who:null}));
  const pool=mine.map(n=>byName[n]).filter(Boolean).sort((a,b)=>b.v-a.v);
  const used=new Set();
  // dedicated slots first, best eligible player into each
  board.forEach(s=>{ if(!s.ok||s.ok.length!==1) return;
    const hit=pool.find(p=>!used.has(p.n)&&s.ok.includes(p.p));
    if(hit){s.who=hit;used.add(hit.n);} });
  // then the flex. All eligible players compete for one slot, so compare raw
  // projected points here — VOR would favour a TE purely because TEs are scarce.
  board.forEach(s=>{ if(!s.ok||s.ok.length===1) return;
    const elig=pool.filter(p=>!used.has(p.n)&&s.ok.includes(p.p))
                   .sort((a,b)=>(b.e+(b.p==="TE"?CFG.flexTE:0))-(a.e+(a.p==="TE"?CFG.flexTE:0)));
    const hit=elig[0];
    if(hit){s.who=hit;used.add(hit.n);} });
  // bench
  board.forEach(s=>{ if(s.ok) return;
    const hit=pool.find(p=>!used.has(p.n));
    if(hit){s.who=hit;used.add(hit.n);} });
  return board;
}
const countPos=p=>mine.filter(n=>byName[n]&&byName[n].p===p).length;

// ---- live board reading ----
// Every player has a board rank (the ESPN sheet where we have it, ADP otherwise).
// As picks come in we measure whether a position is going FASTER than that board
// implies, and pull its expected availability forward accordingly.
// The board the opponents will ACTUALLY use, in priority order: the league's
// own Yahoo player list where we captured it (`y`), the ESPN sheet where we
// have it (`espn`), ADP otherwise. Yahoo is preferred because it is the list
// on screen in front of the other nine managers during the draft.
function boardRank(p){ return p.y || p.espn || p.a; }

function pace(){
  const pick=curPick();
  if(pick<20) return {QB:1,RB:1,WR:1,TE:1,K:1,DST:1};
  const taken={},expected={};
  for(const q of PLAYERS){
    if(q.v<=-90) continue;
    const isGone=gone.has(q.n)||mine.includes(q.n);
    taken[q.p]=(taken[q.p]||0)+(isGone?1:0);
    // how many at this position "should" be gone by now
    expected[q.p]=(expected[q.p]||0)+(boardRank(q)<pick?1:0);
  }
  const out={};
  for(const pos in taken){
    const e=Math.max(expected[pos]||0,1), t=taken[pos]||0;
    out[pos]=Math.max(0.6,Math.min(1.45,t/e));  // >1 hot; capped so a run
                                                // discounts reach, never erases it
  }
  return out;
}

// Probability a player is still there at pick N, given the live pace.
function survives(p,N,pc){
  const hot=pc[p.p]||1;
  const eff=boardRank(p)/hot;                   // hot position => effectively earlier
  const sd=Math.max(4,eff*0.16);
  return 1/(1+Math.exp((N-eff)/sd));
}

// Best VOR at a position I can realistically still get at my NEXT pick.
function expectedBest(pos,N,pc){
  let best=null;
  for(const q of PLAYERS){
    if(q.p!==pos||gone.has(q.n)||mine.includes(q.n)||q.v<=-90) continue;
    if(survives(q,N,pc)>=CFG.survThresh){ if(best===null||q.v>best) best=q.v; }
  }
  return best;
}

// ---- recommendation engine ----
// Additive weighting. Multiplicative weights invert on negative VOR, which is
// what let a 4th QB outrank a startable WR in the late rounds.
const MAX={QB:2,RB:6,WR:8,TE:2,K:1,DST:1};
const REQ={QB:1,RB:2,WR:2,TE:1,K:1,DST:1};   // starters. Flex handled separately.
const FLEXOK=["RB","WR","TE"];

function shortfall(){
  const s={},need={};let total=0;
  for(const p in REQ){ const d=Math.max(0,REQ[p]-countPos(p)); s[p]=d; total+=d; }
  const surplus=FLEXOK.reduce((a,p)=>a+Math.max(0,countPos(p)-REQ[p]),0);
  s.FLEX=surplus>0?0:1; total+=s.FLEX;
  return {s,total};
}
const picksLeft=()=>Math.max(0,ROUNDS-mine.length);
function rosterByes(){
  const c={}; mine.forEach(n=>{const p=byName[n]; if(p) c[p.b]=(c[p.b]||0)+1;}); return c;
}
function byeLoad(){
  const c={}; fillRoster().slice(0,7).forEach(s=>{   // QB,WR,WR,RB,RB,TE,FLEX
    if(s.who) c[s.who.b]=(c[s.who.b]||0)+1; });
  return c;
}

function lateBest(pos,fromPick){
  let best=null;
  for(const q of PLAYERS){
    if(q.p!==pos||gone.has(q.n)||mine.includes(q.n)||q.v<=-90) continue;
    if(boardRank(q) < fromPick+CFG.lateHorizon) continue;  // near-certain to still be there
    if(best===null||q.v>best) best=q.v;
  }
  return best;
}
function scarcity(p,pick){
  const lb=lateBest(p.p,pick);
  if(lb===null) return 45;
  return Math.max(CFG.scarMin, Math.min(CFG.scarMax, p.v-lb));
}
function needBonus(pos,rnd,p_ev_hint,P_,pick_){
  const have=countPos(pos);
  if(have>=MAX[pos]) return null;                     // hard cap: never recommend
  const {s,total}=shortfall(), left=picksLeft();
  const fillsSlot = s[pos]>0;
  const fillsFlex = s.FLEX>0 && FLEXOK.includes(pos) && !fillsSlot
                    && !(pos==="TE"&&have>=1);
  // endgame: if you have exactly enough picks left to fill your empty starting
  // slots, nothing else may be recommended.
  if(left<=total) return (fillsSlot||fillsFlex)?900:null;
  // kicker and defense are streamable; only in the last two rounds
  if(pos==="K")   return rnd>=CFG.kRound?800:null;
  if(pos==="DST") return rnd>=CFG.dstRound?800:null;
  // never a backup QB while real roster work remains
  if(pos==="QB"&&have>=1) return left<=2?-30:null;
  // Under 6-point passing TDs the QB1 is a genuine early-round asset; VOR now
  // carries the timing on its own, so only a mild late nudge remains.
  if(pos==="QB") return (fillsSlot?scarcity(P_,pick_):0) + (rnd<=1?CFG.qbEarly:rnd<=4?CFG.qbMid:0);
  if(pos==="TE"&&have===0) return scarcity(P_,pick_);
  if(fillsSlot) return scarcity(P_,pick_);
  if(fillsFlex){
    // worth the flex only if he'd actually outscore the alternatives there
    const bestFlex=Math.max(0,...mine.map(n=>byName[n]).filter(x=>x&&FLEXOK.includes(x.p)).map(x=>x.e));
    return p_ev_hint>=bestFlex-25?22:6;
  }
  // bench balance: you start 2 WR + a flex, so don't leave yourself 3 deep there
  if(pos==="WR"&&have<4) return CFG.benchWR;
  if(pos==="RB"&&have<4) return CFG.benchRB;
  if(pos==="RB"&&have>=4&&countPos("WR")<5) return -CFG.rb5pen; // a 5th RB over a 4th WR is waste
  return 0;                                            // pure bench upside
}
function bestSurviving(pos,nextPick){
  // best VOR at this position likely to last until my next pick
  let best=-999;
  for(const q of PLAYERS){
    if(q.p!==pos||gone.has(q.n)||mine.includes(q.n)||q.v<=-90) continue;
    if(q.a < nextPick) continue;        // realistically gone by then
    if(q.v>best) best=q.v;
  }
  return best===-999?null:best;
}
let PACE={QB:1,RB:1,WR:1,TE:1,K:1,DST:1};
function score(p,pick,rnd){
  const nb=needBonus(p.p,rnd,p.e,p,pick);
  if(nb===null) return null;
  let s=p.v+nb, why=[];
  const sf=shortfall().s;
  if(nb>=900) why.push("<b>must fill</b> — you have "+picksLeft()+" picks and "+shortfall().total+" empty starting slots");
  else if(nb>=800) why.push("<b>last-round slot</b>");
  else if(sf[p.p]>0) why.push("<b>fills your open "+p.p+"</b> slot");
  else if(sf.FLEX>0&&FLEXOK.includes(p.p)) why.push("fills your <b>flex</b>");
  else why.push("bench value");
  if(p.v>=30) why.push("<b>"+Math.round(p.v)+" pts</b> over a replacement "+p.p);
  const bl=byeLoad(), rl=rosterByes();
  if(FLEXOK.concat("QB").includes(p.p)){
    const st=bl[p.b]||0, all=rl[p.b]||0;
    if(st>=4){ s-=CFG.bye4; why.push("<b>bye disaster</b> — "+st+" starters off in week "+p.b); }
    else if(st>=3){ s-=CFG.bye3; why.push("<b>bye disaster</b> — "+st+" starters already off in week "+p.b); }
    else if(st>=2){ s-=CFG.bye2; why.push("<b>bye clash</b> — you already start "+st+" in week "+p.b); }
    else if(all>=4){ s-=CFG.byeAll; why.push(all+" of your roster is off in week "+p.b); }
  }
  const sameSkill=mine.filter(n=>byName[n]&&byName[n].t===p.t&&FLEXOK.includes(byName[n].p)).length;
  if(sameSkill>=2&&FLEXOK.includes(p.p)){ s-=CFG.conc;
    why.push("third piece of the "+p.t+" offense"); }
  let isCuff=false;
  if(p.p==="RB"){
    const prot=mine.map(n=>byName[n]).filter(x=>x&&x.p==="RB"&&x.t===p.t&&x.v>=40)
                   .sort((a,b)=>b.v-a.v)[0];
    if(prot){ isCuff=true; s+=prot.v>=80?CFG.cuffTop:CFG.cuff;
      why.push("<b>handcuff</b> for "+prot.n+" — your season swings on him"); }
  }
  // Bye coverage. Only counts once your starters at that position are set.
  if(FLEXOK.concat("QB").includes(p.p)){
    const starters=fillRoster().slice(0,7).filter(x=>x.who).map(x=>x.who);
    const sameSlot=starters.filter(x=>x.p===p.p);
    if(sameSlot.length){
      const offWeeks=new Set(sameSlot.map(x=>x.b));
      const covered=mine.map(n=>byName[n]).filter(x=>x&&x.p===p.p&&!sameSlot.includes(x));
      if(offWeeks.has(p.b)){
        s-=CFG.byeNoCover; why.push("<b>no cover</b> — off the same week as your starting "+p.p);
      }else if([...offWeeks].some(w=>!covered.some(c=>c.b!==w))){
        s+=CFG.byeCover; why.push("<b>covers week "+[...offWeeks][0]+"</b>, when your "+p.p+" is off");
      }
    }
  }
  const myQB=mine.map(n=>byName[n]).filter(x=>x&&x.p==="QB");
  if(["WR","TE"].includes(p.p)&&myQB.some(q=>q.t===p.t)){ s+=CFG.stack; why.push("stacks with your "+p.t+" QB"); }
  // OPPORTUNITY COST, pace-aware. What does taking him now buy over the best
  // player at his position I can still expect at my next pick?
  const picksArr=myPicks(), nxt=picksArr.find(x=>x>pick);
  if(nxt&&nb<800){
    const pc=PACE, eb=expectedBest(p.p,nxt,pc);
    if(eb!==null){
      const drop=p.v-eb;
      if(drop>CFG.dropMin){
        s+=Math.min(CFG.dropCap,(drop-CFG.dropMin)*CFG.dropMult);
        why.push("<b>+"+Math.round(drop)+" over the best "+p.p+" expected at pick "+nxt+"</b>");
      }
    }
    const hot=pc[p.p]||1;
    if(hot>=1.35&&p.v>15){
      s+=Math.min(CFG.runCap,(hot-1.35)*CFG.runRate);
      why.push("<b>"+p.p+" run in progress</b> — going "+Math.round((hot-1)*100)+"% faster than the board");
    }
    if(hot<=0.75&&survives(p,nxt,pc)>0.75){
      s-=CFG.coldPen; why.push(p.p+"s are falling; he should last to "+nxt);
    }
  }
  // Reach is measured against where he will ACTUALLY go given the live pace,
  // not against a static preseason board. In a run, "reaching" is just paying
  // the new market price.
  // Only a HOT position pulls a player forward. A cold one does not push him
  // later than his board rank already implies, so clamp at 1.0.
  const hotNow=Math.max(1.0, PACE[p.p]||1);
  const effRank=boardRank(p)/hotNow;
  const reach=boardRank(p)-pick;   // static board; pace no longer discounts reach
  if(reach>CFG.reachFree&&nb<800){
    // insurance on your own star carries option value ADP does not price, so a
    // handcuff is not penalised for being "early".
    s-=(reach-CFG.reachFree)*(isCuff?CFG.cuffMult:CFG.reachRate);
    if(reach>18) why.push("<b>"+Math.round(reach)+" picks early</b> — he will still be there");
    else why.push("likely still there next round");
  }
  if(reach<-6) why.push("<b>falling</b> — ADP "+Math.round(p.a));
  if(p.r===1&&p.av<1){
    why.push("<b>availability "+Math.round(p.av*100)+"%</b> — EV already cut for it: "+
             p.d.split(".")[0].toLowerCase());
  }
  if(p.src==="E"){   // estimated from ADP; the ruleset effect is not yet in his EV
    if(p.s<=-3) why.push("our scoring adds value here");
    if(p.s>=4)  why.push("our scoring <b>costs</b> him vs standard");
  }
  return {p,s,why};
}
// ============================================================================
// FORWARD PLANNING — what TEAM does this pick lead to?
//
// The scorer prices one pick, with a one-step glance at what survives to the
// next. That is not how a draft is won. Winning is a ROUTE: take the receiver
// now because two of these three backs will still be here at 26 and none of
// these receivers will.
//
// planRoute() drafts the REST of my draft for me. At every remaining pick it
// takes whoever these same rules would take, from the players expected to still
// be on the board by then, and hands back the team that results. Candidates are
// then judged on that team rather than on themselves.
//
// The measure of a team is not the sum of its players. It is what it puts on
// the field WEEK BY WEEK: on a bye week the starter is replaced by whoever is
// left, and a roster whose byes collide loses real points that a season total
// never shows. seasonPoints() adds up all seventeen weeks, so complementary
// byes and useful depth are worth exactly what they are worth, and nothing is
// double-counted by a hand-tuned bonus.
// ============================================================================
const WEEKS = 17;

// Virtual-roster versions of the roster rules. Deliberately separate from the
// live ones, which read globals: a refactor of the shipped scorer is not
// something to attempt on the afternoon of a draft.
function vCount(r, pos) { let n = 0; for (const p of r) if (p.p === pos) n++; return n; }
function vShort(r) {
  const s = {}; let total = 0;
  for (const pos in REQ) { const d = Math.max(0, REQ[pos] - vCount(r, pos)); s[pos] = d; total += d; }
  const surplus = FLEXOK.reduce((a, pos) => a + Math.max(0, vCount(r, pos) - REQ[pos]), 0);
  s.FLEX = surplus > 0 ? 0 : 1; total += s.FLEX;
  return { s: s, total: total };
}
function vNeed(r, pos, rnd, left) {
  if (vCount(r, pos) >= MAX[pos]) return null;
  const sh = vShort(r), fillsSlot = sh.s[pos] > 0;
  const fillsFlex = sh.s.FLEX > 0 && FLEXOK.indexOf(pos) >= 0 && !fillsSlot &&
                    !(pos === "TE" && vCount(r, pos) >= 1);
  if (left <= sh.total) return (fillsSlot || fillsFlex) ? 900 : null;
  if (pos === "K") return rnd >= CFG.kRound ? 800 : null;
  if (pos === "DST") return rnd >= CFG.dstRound ? 800 : null;
  if (pos === "QB" && vCount(r, "QB") >= 1) return left <= 2 ? -30 : null;
  if (fillsSlot || fillsFlex) return 40;
  if (pos === "WR" && vCount(r, pos) < 4) return CFG.benchWR;
  if (pos === "RB" && vCount(r, pos) < 4) return CFG.benchRB;
  if (pos === "RB" && vCount(r, pos) >= 4 && vCount(r, "WR") < 5) return -CFG.rb5pen;
  return 0;
}

// The starting lineup for one week, given who is not on a bye.
function vLineup(r, week) {
  const board = SLOTS.filter(x => x[1]);           // starting slots only
  const pool = r.filter(p => p.b !== week).sort((a, b) => b.e - a.e);
  const used = {}; const out = [];
  for (const sl of board) {
    const ok = sl[1];
    if (ok.length !== 1) continue;                 // flex resolved after
    const hit = pool.find(p => !used[p.n] && ok.indexOf(p.p) >= 0);
    if (hit) { used[hit.n] = 1; out.push(hit); }
  }
  for (const sl of board) {
    const ok = sl[1];
    if (ok.length === 1) continue;
    const hit = pool.filter(p => !used[p.n] && ok.indexOf(p.p) >= 0)
                    .sort((a, b) => (b.e + (b.p === "TE" ? CFG.flexTE : 0)) -
                                    (a.e + (a.p === "TE" ? CFG.flexTE : 0)))[0];
    if (hit) { used[hit.n] = 1; out.push(hit); }
  }
  return out;
}
// What this roster actually scores, week by week, over the season. An empty
// starting slot scores nothing -- which is exactly the cost of a bye you did
// not cover, and the reason two studs sharing a bye are worth less than the
// same two points spread across different weeks.
function seasonPoints(r) {
  let tot = 0;
  for (let w = 1; w <= WEEKS; w++) for (const p of vLineup(r, w)) tot += p.e / WEEKS;
  return tot;
}

// Everyone still expected to be on the board at pick N. Only the top few of
// each position are returned: the rollout evaluates a whole season for every
// candidate it considers, and the fortieth-best receiver is not going to be
// the answer. This is the difference between a plan that arrives inside the
// pick clock and one that does not.
const ROLL_WIDTH = 10;
function expectedAvail(N, pc, taken) {
  const byPos = {};
  for (const q of PLAYERS) {
    if (q.v <= -90 || gone.has(q.n) || mine.indexOf(q.n) >= 0 || taken[q.n]) continue;
    if (survives(q, N, pc) < CFG.survThresh) continue;
    (byPos[q.p] = byPos[q.p] || []).push(q);
  }
  const out = [];
  for (const pos in byPos) {
    byPos[pos].sort((a, b) => b.e - a.e);
    for (let i = 0; i < Math.min(ROLL_WIDTH, byPos[pos].length); i++) out.push(byPos[pos][i]);
  }
  return out;
}

// Draft the rest of my picks, assuming the board behaves as expected.
function planRoute(first, pick, pc) {
  const taken = {};
  const roster = mine.map(n => byName[n]).filter(Boolean);
  const route = [];
  if (first) { taken[first.n] = 1; roster.push(first); route.push({ pick: pick, p: first }); }
  const rest = myPicks().filter(x => x > pick);
  for (let i = 0; i < rest.length; i++) {
    const N = rest[i], after = rest[i + 1] || null;
    if (roster.length >= ROUNDS) break;
    const rnd = Math.min(ROUNDS, Math.ceil(N / TEAMS));
    const left = ROUNDS - roster.length;
    const avail = expectedAvail(N, pc, taken);
    const base = seasonPoints(roster);

    // What taking a man NOW buys over waiting one turn at his position. Greedy
    // on lineup points alone drafted Bo Nix at pick 15 -- a quarterback scores
    // the most raw points, so he always looked like the biggest addition. But
    // every team ends up with a quarterback, and Nix goes at 113. What matters
    // is the gap between him and the best of his position still there NEXT
    // time, which is replacement value measured in actual lineup points rather
    // than assumed from a table.
    const gainOf = q => { roster.push(q); const g = seasonPoints(roster); roster.pop(); return g - base; };
    const wait = {};
    if (after) {
      const later = expectedAvail(after, pc, taken);
      for (const pos in MAX) {
        let bestLater = null;
        for (const q of later) if (q.p === pos && (!bestLater || q.e > bestLater.e)) bestLater = q;
        wait[pos] = bestLater ? gainOf(bestLater) : 0;
      }
    }

    let best = null, bs = -1e9;
    for (const q of avail) {
      const nb = vNeed(roster, q.p, rnd, left);
      if (nb === null) continue;
      // seasonPoints already prices the bye clash, the depth and the flex, so
      // nothing here needs a hand-tuned bonus and nothing can double-count.
      const sc = gainOf(q) - (wait[q.p] || 0) + (nb >= 800 ? nb : 0);
      // Ties are common and must not fall to whoever happens to sit earlier in
      // the array: when the best of a position is expected to survive anyway,
      // every candidate there scores about zero, and array order was quietly
      // choosing between them.
      if (sc > bs + 1e-9 || (best && Math.abs(sc - bs) <= 1e-9 && q.e > best.e)) { bs = sc; best = q; }
    }
    if (!best) break;
    taken[best.n] = 1; roster.push(best); route.push({ pick: N, p: best });
  }
  return { route: route, roster: roster, pts: seasonPoints(roster) };
}

// The route, on the card. "Take him and this is the team you finish with" is
// the whole point of planning ahead, and it was being computed and thrown away.
// Only on the top card: five of these is noise, one is a plan.
function shortName(p){
  if(p.p==="DST") return p.n.split(" ")[0];
  const t=p.n.replace(/\s+(Jr\.|Sr\.|III|II|IV|V)$/,"").split(" ");
  return t.length>1?t[t.length-1]:t[0];
}
function routeLine(r){
  if(!r.route||!r.route.length) return "";
  const seg=r.route.slice(0,6).map(x=>'<b>'+x.pick+'</b> '+shortName(x.p)).join(" &rsaquo; ");
  return '<div class="route"><span class="rk">then</span>'+seg+
    (r.route.length>6?" &hellip;":"")+'</div>';
}

function recommend(){
  const pick=curPick(), rnd=Math.min(ROUNDS,Math.ceil(pick/TEAMS));
  PACE=pace();
  const live=PLAYERS.filter(p=>!gone.has(p.n)&&!mine.includes(p.n)&&p.v>-90);
  let out=live.map(p=>score(p,pick,rnd)).filter(Boolean).sort((a,b)=>b.s-a.s).slice(0,8);
  if(out.length){
    if(!PLAN) return out.slice(0,5);
    // render() runs on every board change, and several times per change. The
    // rollout is the expensive part of the tool by two orders of magnitude, so
    // it is computed once per actual board state and then reused.
    const sig=pick+"|"+gone.size+"|"+mine.length+"|"+slot+"|"+out.map(c=>c.p.n).join(",");
    if(planCache.sig===sig) return planCache.out;
    // Re-rank the shortlist on the team each pick leads to. The per-pick score
    // chose the shortlist -- it knows about runs, reaches and handcuffs, which
    // a rollout cannot see -- and the route then decides the order, because the
    // season is won by the roster you finish with, not the pick you enjoy most.
    for(const c of out){
      const pl=planRoute(c.p,pick,PACE);
      c.route=pl.route.slice(1); c.pts=pl.pts;
    }
    const bestPts=Math.max(...out.map(c=>c.pts));
    for(const c of out){
      c.lost=bestPts-c.pts;
      if(c.lost>0.5) c.why.push("costs <b>"+c.lost.toFixed(0)+" pts</b> of finished team vs the best route");
      else c.why.unshift("<b>best finished team</b> — "+Math.round(c.pts)+" starting pts");
    }
    out.sort((a,b)=>b.pts-a.pts||b.s-a.s);
    planCache={sig:sig,out:out.slice(0,5)};
    return planCache.out;
  }
  // FALLBACK 1: a required position exists but the pool is empty there.
  const {s:sf}=shortfall();
  const missing=Object.keys(sf).filter(k=>k!=="FLEX"&&sf[k]>0);
  if(missing.length){
    const anyLeft=live.filter(p=>missing.includes(p.p));
    if(anyLeft.length){
      return anyLeft.sort((a,b)=>b.v-a.v).slice(0,5).map(p=>({p,s:p.v,
        why:["<b>fills your open "+p.p+"</b> slot"]}));
    }
    exhausted=missing;   // nothing at that position remains in the pool at all
  }
  // FALLBACK 2: take the best player left rather than showing nothing.
  return live.sort((a,b)=>b.v-a.v).slice(0,5).map(p=>({p,s:p.v,
    why:["<b>best available</b> — no required-slot player left in the pool"]}));
}

// ---- mock draft ----
// Opponents are modelled as experienced, rational drafters: they draft on value
// over replacement rather than raw ADP, they respect roster construction, and
// they reach far less than the field. ADJUST controls how far they have
// recalibrated to OUR ruleset (6-pt passing TDs, sack penalty):
//   0   = sharp drafters still priced to standard scoring
//   0.5 = they have noticed and half-corrected      <-- default
//   1   = they have fully repriced; our QB edge is gone
let ADJUST=0.5;

function gauss(){ let u=0,v=0; while(!u)u=Math.random(); while(!v)v=Math.random();
  return Math.sqrt(-2*Math.log(u))*Math.cos(2*Math.PI*v); }

const OREPL={};
(function(){ const r={QB:10,RB:24,WR:28,TE:10,K:10,DST:10};
  for(const pos in r){ const g=PLAYERS.filter(p=>p.p===pos&&p.v>-90).sort((a,b)=>b.e-a.e);
    OREPL[pos]=g[Math.min(r[pos],g.length)-1].e; } })();

// what an opponent thinks a player is worth
const OREPL_STD={};
(function(){ const r={QB:10,RB:24,WR:28,TE:10,K:10,DST:10};
  for(const pos in r){ const g=PLAYERS.filter(p=>p.p===pos&&p.v>-90)
      .map(p=>p.e-(p.s||0)).sort((a,b)=>b-a);
    OREPL_STD[pos]=g[Math.min(r[pos],g.length)-1]; } })();
function oppValue(p){
  const ev   = p.e - (1-ADJUST)*(p.s||0);
  const base = OREPL[p.p]*ADJUST + OREPL_STD[p.p]*(1-ADJUST);
  let v = ev - base;
  if(p.p==="K"||p.p==="DST") v*=0.2;
  return v;
}
const OMAX={QB:2,RB:6,WR:7,TE:2,K:1,DST:1};
const OREQ={QB:1,RB:2,WR:2,TE:1,K:1,DST:1};
function oppNeed(pos,have,rnd,left){
  if((have[pos]||0)>=OMAX[pos]) return null;
  let short=0; for(const k in OREQ) short+=Math.max(0,OREQ[k]-(have[k]||0));
  const fills=(have[pos]||0)<OREQ[pos];
  if(left<=short) return fills?900:null;            // must fill starters
  if(pos==="K")   return rnd>=14?800:null;
  if(pos==="DST") return rnd>=13?800:null;
  if(pos==="QB"&&(have.QB||0)>=1) return left<=2?-40:null;
  if(pos==="TE"&&(have.TE||0)===0&&rnd>=8) return 45;
  return fills?40:0;
}

// NAME MATCHING
// Two callers, one matcher:
//   - the search box, where you type "stfd" and press Enter;
//   - the Yahoo bridge, which reads a name off the draft board and must decide
//     whether it is confident enough to commit it without asking.
// Both need the same normalisation, so both use these functions. The bridge
// additionally requires a confidence score, because "Robinson" alone is three
// different players and committing the wrong one silently corrupts the board.
// ============================================================================
const SUFFIX=/\b(?:jr|sr|ii|iii|iv)\b/g;
function norm(s){
  return String(s==null?"":s).toLowerCase()
    .normalize("NFD").replace(/[\u0300-\u036f]/g,"")   // strip accents
    .replace(/[^a-z0-9]+/g," ")                       // ' . - all become spaces
    .replace(SUFFIX," ")                              // Jr / Sr / III
    .replace(/\s+/g," ").trim();
}
// Precomputed once. 237 players; every keystroke scores all of them.
const NIDX=PLAYERS.map(p=>{
  const tok=norm(p.n).split(" ").filter(Boolean);
  return {p, tok, flat:tok.join(""), last:tok[tok.length-1]||"",
          first:tok[0]||"", init:tok.map(t=>t[0]).join(""), team:norm(p.t)};
});
const NIDX_BY_NAME=Object.fromEntries(NIDX.map(x=>[x.p.n,x]));

// Is q a subsequence of s, and how tight a one? 0 = no match, higher = better.
function subseq(q,s){
  let i=0,first=-1,prev=-2,gaps=0;
  for(let k=0;k<s.length&&i<q.length;k++){
    if(s.charCodeAt(k)===q.charCodeAt(i)){
      if(first<0)first=k;
      if(k!==prev+1)gaps++;
      prev=k;i++;
    }
  }
  if(i<q.length) return 0;
  const span=prev-first+1;
  return 1/(1+(span-q.length)*0.6+first*0.35+Math.max(0,gaps-1)*0.4);
}
// Every query token must prefix a distinct name token, left to right.
// "matt staf" -> Matthew Stafford; "ja ch" -> Ja'Marr Chase.
function tokenPrefix(qt,nt){
  let ni=0,tight=0;
  for(const q of qt){
    let hit=-1;
    for(let k=ni;k<nt.length;k++) if(nt[k].startsWith(q)){hit=k;break;}
    if(hit<0) return 0;
    tight+=q.length/nt[hit].length;
    ni=hit+1;
  }
  return tight/qt.length;
}
// Score one player against a raw query string. Additive-free: this is a plain
// max over strategies, so a strong exact match can never be diluted by a weak
// fuzzy one.
function nameScore(q,x){
  const qn=norm(q); if(!qn) return 0;
  const qt=qn.split(" ").filter(Boolean);
  // Spaces in the query must not have to line up with spaces in the name:
  // "aj brown" and "ajbrown" both have to reach "A.J. Brown" (-> "a j brown").
  const qflat=qt.join("");
  const idx=x.tok?x:NIDX_BY_NAME[x.n||x];
  if(!idx) return 0;
  let s=0;
  const bump=v=>{ if(v>s) s=v; };
  if(qflat===idx.flat||qn===idx.tok.join(" ")) bump(1000);
  if(qt.length>1){ const t=tokenPrefix(qt,idx.tok); if(t) bump(780+60*t); }
  if(idx.init===qn&&qn.length>=2) bump(760);                 // "jsn"
  if(idx.last===qn) bump(700);
  if(idx.last.startsWith(qn)) bump(600+40*(qn.length/idx.last.length));
  if(idx.first.startsWith(qn)) bump(520+40*(qn.length/idx.first.length));
  if(idx.init.startsWith(qn)&&qn.length>=2) bump(450);
  if(idx.team===qn) bump(430);                               // "sea" -> Seattle
  const sl=subseq(qflat,idx.last); if(sl) bump(300+120*sl);   // "stfd"
  const sf=subseq(qflat,idx.flat); if(sf) bump(200+120*sf);
  return s;
}
// Ranked search over the pool. Gone/mine players are included but flagged, so
// you can see that a name you typed is already off the board.
function searchPlayers(q,opts){
  const o=opts||{}, lim=o.limit||8;
  if(!norm(q)) return [];
  const out=[];
  for(const idx of NIDX){
    const isGone=gone.has(idx.p.n), isMine=mine.includes(idx.p.n);
    if(o.available&&(isGone||isMine)) continue;
    const s=nameScore(q,idx);
    if(s>0) out.push({p:idx.p,s,gone:isGone,mine:isMine});
  }
  // VOR breaks ties: two equally-good string matches, the better player first.
  out.sort((a,b)=>b.s-a.s||b.p.v-a.p.v);
  return out.slice(0,lim);
}

// Match a name read off the Yahoo board. Returns a confidence in 0..1.
// A raw board cell is usually noisier than a typed query — "M. Stafford LAR - QB"
// — so the position/team hints are used as corroboration, never as the match.
function matchBoardName(raw,hint){
  const h=hint||{};
  const cand=searchPlayers(raw,{limit:8});
  if(!cand.length) return {p:null,conf:0,why:"no candidate"};
  const best=cand[0].s;
  // Everything within a hair of the best string score is genuinely ambiguous on
  // the name alone — "Robinson" is three players. The team and position printed
  // in the same board cell must therefore CHOOSE among them, not merely score
  // the choice after the fact. Corroboration is confined to this tied group so
  // it can never promote a worse string match over a better one.
  const tied=cand.filter(c=>best-c.s<=60);
  const corr=c=>{
    let k=0;
    if(h.pos){ k+= c.p.p===h.pos?2:-3; }
    if(h.team&&norm(h.team).length>1){ k+= norm(h.team)===norm(c.p.t)?2:-3; }
    return k;
  };
  tied.sort((a,b)=>corr(b)-corr(a)||b.s-a.s||b.p.v-a.p.v);
  const top=tied[0], runner=tied[1];
  const sep=runner?corr(top)-corr(runner):99;   // did the hints actually separate them?

  let conf;
  if(top.s>=1000) conf=1;            // exact full name
  else if(top.s>=760) conf=0.92;     // initials, or every token prefixed
  else if(top.s>=600) conf=0.75;     // surname only
  else if(top.s>=430) conf=0.55;
  else conf=0.35;
  // Ambiguous on the name and NOT separated by the hints: this is the case that
  // silently corrupts the board, so it must never clear the auto-commit bar.
  // Two men can be identical on name, position AND team: Bijan Robinson and
  // Brian Robinson are both Atlanta running backs, so "B. ROBINSON RB ATL"
  // describes them equally well and the hints separate nothing. What separates
  // them is that one goes first overall and the other goes undrafted. A board
  // rank that far apart IS separation, and refusing to use it left the best
  // player in the draft sitting in the confirm strip waiting for a tap, every
  // single time.
  const gulf = runner ? Math.abs(boardRank(runner.p)-boardRank(top.p)) : 999;
  if(tied.length>1&&sep<=0&&gulf<60) conf=Math.min(conf,0.5);
  else if(tied.length>1&&sep<=0) conf=Math.min(conf,0.9);   // separated, but not by name
  // An EXACT, unique full-name match identifies a player outright in a pool of
  // 237. If Yahoo then disagrees about his team, by far the likeliest
  // explanation is that our team field is stale — players change teams every
  // offseason — not that this is a different man with the same full name. So a
  // team disagreement is reported rather than treated as a failed match.
  // Position is far more stable than team, so a position clash still demands a
  // tap even on an exact name.
  const exactUnique = top.s>=1000 && tied.length===1;
  let stale=null;
  if(h.pos) conf = top.p.p===h.pos ? Math.min(1,conf+0.12)
                                   : Math.min(conf, exactUnique?0.8:0.4);
  if(h.team&&norm(h.team).length>1){
    if(norm(h.team)===norm(top.p.t)) conf=Math.min(1,conf+0.12);
    else if(exactUnique){ conf=Math.min(conf,0.9); stale="board says "+h.team+", our data says "+top.p.t; }
    else conf=Math.min(conf,0.45);
  }
  if(top.gone||top.mine) return {p:top.p,conf,why:"already off the board",dup:true,stale};
  return {p:top.p,conf,why:"matched",alts:tied.slice(1,4).map(c=>c.p),stale};
}

// ---- drop-off view ----
// The engine already computes what the best player at each position will be at
// your NEXT pick — it is how opportunity cost is priced. It was only ever used
// internally. Surfacing it lets you see the trade-off yourself instead of
// taking the recommendation on trust: "RB 92 -> 61" means waiting a round costs
// 31 points of VOR at running back, and nothing at a position showing 0.
// Read-only. It computes nothing the engine was not already computing.
function dropOff(){
  const pick=curPick(), nxt=myPicks().find(x=>x>pick);
  if(!nxt) return [];
  const pc=pace(), out=[];
  for(const pos of ["QB","RB","WR","TE"]){
    let now=null;
    for(const q of PLAYERS){
      if(q.p!==pos||gone.has(q.n)||mine.includes(q.n)||q.v<=-90) continue;
      if(now===null||q.v>now) now=q.v;
    }
    if(now===null) continue;
    const eb=expectedBest(pos,nxt,pc);
    out.push({pos,now,next:eb,drop:eb===null?null:now-eb});
  }
  // Steepest cliff first: that is the position you cannot afford to wait on.
  return out.sort((a,b)=>(b.drop==null?-1:b.drop)-(a.drop==null?-1:a.drop));
}

// ---- persistence ----
// This used to call window.storage, which is not a browser API. Opened as a
// local file the call threw, the throw was swallowed by the catch, and NOTHING
// WAS EVER SAVED — one accidental refresh mid-draft and the board was gone.
// localStorage is synchronous, needs no server, and survives a reload, which is
// exactly the offline guarantee the tool is supposed to have.
const LSKEY="sstlv-live-v2";
// Yahoo mock drafts and the real draft room share one origin, so they shared
// one saved board: rehearse in a mock, then open the real draft, and the tool
// restores the MOCK's picks, roster and pick counter -- silently, because that
// is exactly what restoring is supposed to look like. Keying the save by draft
// keeps them apart and lets you move between them without losing either.
//
// The id is the longest 5+ digit run in the URL, which is the league number in
// both of Yahoo's draft-room URL shapes; the offline file falls back to its
// path. Getting it wrong costs an empty board that RE-IMPORT refills, which is
// a great deal better than a board full of somebody else's mock draft.
function draftId(){
  try{
    const runs=String(location.href||"").match(/\d{5,}/g);
    if(runs&&runs.length) return "d"+runs.sort((a,b)=>b.length-a.length)[0];
    return "p"+String(location.pathname||"file").replace(/[^a-z0-9]+/gi,"").slice(-20);
  }catch(e){ return "file"; }
}
const DRAFT_ID=draftId();
const SKEY=LSKEY+":"+DRAFT_ID;
function store(){
  try{ if(typeof localStorage!=="undefined"&&localStorage) return localStorage; }catch(e){}
  return null;   // private mode / file:// restrictions in some browsers
}
let saveWarned=false;
function save(){
  const st=store();
  if(!st){ if(!saveWarned){ saveWarned=true; console.warn("no localStorage: state will not survive a reload"); } return; }
  try{
    st.setItem(SKEY,JSON.stringify({slot,gone:[...gone],mine,takenAt,goneAt,sortMode,feed,boardPick,v:2}));
  }catch(e){ if(!saveWarned){ saveWarned=true; console.warn("save failed:",e); } }
}
function load(){
  const st=store();
  try{
    let raw=st&&st.getItem(SKEY);
    if(!raw&&st){
      // One-time migration off the un-namespaced key, so upgrading the script
      // MID-DRAFT does not throw the live board away. It is CONSUMED, not
      // copied: the next draft you open starts clean instead of inheriting
      // this one, which is the whole point of namespacing.
      const legacy=st.getItem(LSKEY);
      if(legacy){ raw=legacy; try{ st.setItem(SKEY,legacy); st.removeItem(LSKEY); }catch(e){} }
    }
    if(raw){
      const d=JSON.parse(raw);
      slot=d.slot||null; gone=new Set(d.gone||[]); mine=d.mine||[];
      takenAt=d.takenAt||{}; goneAt=d.goneAt||{}; sortMode=d.sortMode||"value";
      feed=Array.isArray(d.feed)?d.feed:[]; boardPick=d.boardPick==null?null:d.boardPick;
    }
  }catch(e){ console.warn("could not restore saved draft:",e); }
  render();
}
function storageWorks(){ return !!store(); }
