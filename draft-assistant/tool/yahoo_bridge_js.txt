// ============================================================================
// YAHOO DRAFT BOARD BRIDGE
//
// Reads picks off the live Yahoo draft room and feeds them to the engine, so
// no player name is ever typed.
//
// WHY THIS DOES NOT PARSE YAHOO'S MARKUP
// I could not open a live Yahoo draft room while writing this, so any CSS
// selector I hardcoded would be a guess that fails silently on draft night —
// the worst possible failure, because a silent board is indistinguishable from
// a slow one. Instead the reader is built on the one thing that cannot change:
// WE ALREADY KNOW ALL 237 NAMES. It scans text for names from our own pool and
// watches which ones APPEAR. That works whether Yahoo renders a table, a list,
// or a virtualised grid, and it survives a redesign.
//
// THE ONE HARD PROBLEM
// The draft room shows two lists of players: the picks already made, and the
// players still available. Scanning the whole page would mark the available
// list as drafted — every player, instantly. The reader therefore never scans
// the whole page. It scans ONE container that you bind with a single click,
// and it cross-checks that container's behaviour: a pick log only ever GAINS
// names. If a scan produces a burst of additions, it stops and asks rather
// than committing, because a burst means the wrong container.
// ============================================================================

const BR = {
  el: null,            // the bound container
  path: null,          // a re-findable path to it, so a reload rebinds itself
  seen: new Set(),     // pool names already read out of it
  pending: [],         // low-confidence reads waiting on a tap
  armed: false,        // committing, as opposed to just watching
  lastCount: 0,
  paused: false,
  note: "",
};
const BR_KEY = "sstlv-bridge-v1";
const BURST_LIMIT = 5;   // more additions than this in one scan = wrong region

// ---- locating the container again after a reload ----
function elPath(el) {
  const parts = [];
  for (let n = el; n && n.nodeType === 1 && n !== document.body; n = n.parentElement) {
    const par = n.parentElement;
    if (!par) break;
    parts.unshift(Array.prototype.indexOf.call(par.children, n));
  }
  return parts.join("/");
}
function elFromPath(path) {
  if (!path) return null;
  let n = document.body;
  for (const i of path.split("/")) {
    if (!n) return null;
    n = n.children[+i];
  }
  return n || null;
}

// ---- reading names out of the bound container ----
// Yahoo prints a pick as something like "1.06  Bijan Robinson  ATL - RB". We
// take each row's whole text and let the matcher deal with the noise, because
// the matcher already handles abbreviations, punctuation and suffixes.
// A round.pick token is only trusted at the START of a row. Anywhere else a
// "12.5" is far more likely to be projected points than round 12 pick 5, and a
// bogus pick number moves the whole counter.
const RE_PICK_DOT = /^\(?(\d{1,2})\.(\d{1,2})\)?(?!\d)/;  // "3.07 Bijan Robinson"
const RE_PICK_WORD = /\bpick\s*#?\s*(\d{1,3})\b/i;        // Pick 27
// Yahoo's own sidebar numbers picks with a BARE INTEGER at the start of the
// row -- "2  J. GIBBS" -- not round.pick. Without this the reader finds no
// pick numbers at all, which is what makes container resolution fall back to
// its weakest heuristic and bind the wrong element.
const RE_PICK_NUM = /^\s*(\d{1,3})(?![\d.:])/;
// CASE-INSENSITIVE. Yahoo renders these title-case -- "RB • Det • Bye 6" --
// so an uppercase-only pattern stripped the position but left the team behind
// as a residual "name". "Ind" then matched the Indianapolis defence.
// The (?!\.) matters more than it looks: without it the "K" in "K. WALKER III"
// is read as the kicker position, the initial is eaten, and the real position
// later in the row survives into the name.
const RE_POS = /\b(QB|RB|WR|TE|K|DEF|DST|D\/ST)\b(?!\.)/i;
const RE_POS_G = new RegExp(RE_POS.source, "gi");
const RE_TEAM = /\b(ARI|ATL|BAL|BUF|CAR|CHI|CIN|CLE|DAL|DEN|DET|GB|HOU|IND|JAX|KC|LAC|LAR|LV|MIA|MIN|NE|NO|NYG|NYJ|PHI|PIT|SEA|SF|TB|TEN|WAS)\b/i;
const RE_TEAM_G = new RegExp(RE_TEAM.source, "gi");
// Yahoo hangs a status badge off the name -- "P. NACUA [Q]" -- in its own
// element, so it lands at the end of the name after everything else is
// stripped, and "P. NACUA Q" matched NOTHING. Only trailing badges are
// removed: a leading single letter is an initial, not a status.
const RE_BADGE = /(?:\s+(?:Q|O|D|P|IR|SUS|PUP|GTD|DTD|NA|NFI))+$/i;

// textContent glues adjacent elements together: "<span>1.01</span>Bijan" reads
// as "1.01Bijan", which matches neither a pick number nor a name. Walking the
// text nodes and joining on element boundaries restores the separation the row
// visually has.
function rowText(el) {
  let out = "";
  (function walk(n) {
    if (n.nodeType === 3) { out += " " + n.nodeValue; return; }
    if (n.nodeType !== 1) return;
    for (let c = n.firstChild; c; c = c.nextSibling) walk(c);
    out += " ";
  })(el);
  return out.replace(/\s+/g, " ").trim();
}

function rowsOf(container) {
  // Prefer real row-ish children; fall back to the container itself.
  const kids = container.querySelectorAll("li,tr,[role=row],div");
  const out = [];
  const seenText = new Set();
  const push = el => {
    const t = rowText(el);
    if (!t || t.length > 160) return;          // a huge blob is a wrapper, not a row
    if (seenText.has(t)) return;
    seenText.add(t);
    out.push({ el, text: t });
  };
  if (kids.length) kids.forEach(push); else push(container);
  return out;
}

// What is left after stripping pick number, position, team and bye must still
// look like a person's name: at least one token of 3+ letters that is not
// itself a team code. This is what rejects "Ind", "• SF •" and bare numbers.
const MIN_READ = 0.5;
function plausibleName(name) {
  if (!name || name.length < 3) return false;
  const toks = name.split(/[^A-Za-z']+/).filter(Boolean);
  return toks.some(t => t.length >= 3 && !RE_TEAM.test(t));
}

function parseRow(text) {
  let pick = null;
  const d = text.match(RE_PICK_DOT);
  // 10-team snake: round 1-15, slot 1-10. Anything outside that is not a pick.
  if (d && +d[1] >= 1 && +d[1] <= 15 && +d[2] >= 1 && +d[2] <= 10) {
    pick = (+d[1] - 1) * 10 + (+d[2]);
  } else {
    const w = text.match(RE_PICK_WORD);
    if (w && +w[1] >= 1 && +w[1] <= 150) pick = +w[1];
    else {
      const n = text.match(RE_PICK_NUM);
      if (n && +n[1] >= 1 && +n[1] <= 150) pick = +n[1];
    }
  }
  const pm = text.match(RE_POS);
  let pos = pm ? pm[1] : null;
  if (pos === "DEF" || pos === "D/ST") pos = "DST";
  const tm = text.match(RE_TEAM);
  // Strip the tokens the matcher should not see as part of the name.
  const name = text
    .replace(RE_PICK_DOT, " ").replace(RE_PICK_WORD, " ").replace(RE_PICK_NUM, " ")
    .replace(RE_POS_G, " ").replace(RE_TEAM_G, " ")
    .replace(/\b(bye|adp|rank|pts|proj|round)\b.*$/i, " ")
    // Separators are not name characters, and they were what stopped the badge
    // from being stripped: "P. NACUA Q \u2022 \u2022" does not end in a badge, so
    // the whole row failed to match while the bare name next to it succeeded --
    // taking the pick NUMBER out of the read with it.
    .replace(/[\u2022\u00B7|\u2013\u2014,()\[\]]+/g, " ")
    .replace(/\s+/g, " ").trim()
    .replace(RE_BADGE, "").trim();
  return { pick, pos, team: tm ? tm[1] : null, name };
}

// Every pool player currently readable inside `el`. Pure: no state touched.
function readRegion(el) {
  const out = [];
  const dedupe = new Set();
  for (const row of rowsOf(el)) {
    const info = parseRow(row.text);
    if (!plausibleName(info.name)) continue;
    const m = matchBoardName(info.name, { pos: info.pos, team: info.team });
    if (!m.p || dedupe.has(m.p.n)) continue;
    // A defence must be announced as one. Without this, a leftover team code
    // or city in a meta line ("• Ind •") matches that team's DST.
    if (m.p.p === "DST" && !/\b(DEF|DST|D\/ST)\b/i.test(row.text)) continue;
    // Below this the read is noise -- UI labels, ad copy. It is not worth a
    // confirmation tap, so it is not queued at all.
    if (m.conf < MIN_READ) continue;
    // A manager's name sits between the picks and IS a person's name: "jeff"
    // fuzzy-matched Justin Jefferson at 75% and took his slot on the board.
    // Every pick row carries a position or a team; a manager row carries
    // neither. Where neither is present, demand a confident match on a name
    // with two parts to it -- which "jeff", "Landon" and "Team 3" are not.
    if (!info.pos && !info.team &&
        (m.conf < 0.9 || info.name.split(/\s+/).filter(t => t.length > 1).length < 2)) continue;
    dedupe.add(m.p.n);
    out.push({ player: m.p, conf: m.conf, pick: info.pick, alts: m.alts || [],
               raw: row.text, stale: m.stale || null });
  }
  return out;
}

// You will click a player's NAME, or the team abbreviation next to it, or the
// whitespace in a row — never the list element itself. Binding the exact node
// under the cursor therefore binds a leaf that holds one field of one row, and
// the reader sees nothing. So walk up from whatever was clicked and take the
// ancestor that actually looks like the list.
// Climbing must also know when to STOP. Climbing while the count keeps rising
// walks straight past the pick list into whatever wrapper holds the pick list
// AND the available list — which then imports the entire player pool as drafted.
//
// The discriminator is the pick number. A pick log prints one per row; an
// available-players list does not. So the target is the SMALLEST ancestor that
// sees the most pick-numbered rows: going higher adds names but no new pick
// numbers, which is exactly the signal that we have left the log.
const MAX_REGION = 60;
// How far to climb. This was 8, which sounds generous and is not: the live room
// puts SIX styled wrappers between the pick row and the list, so a click on the
// "RB - Ind - Bye 13" line ran out of levels while still inside one pick, every
// level read zero players, and the bind landed on that line -- "bound", empty,
// silent. Climbing is cheap because MAX_SUBTREE stops it dead at the first
// wrapper big enough to hold the available-players list.
const MAX_CLIMB = 16;
const MAX_SUBTREE = 1200;      // elements; above this we are out of the log
function resolveContainer(el) {
  const levels = [];
  let n = el;
  for (let up = 0; up < MAX_CLIMB && n && n !== document.body; up++) {
    if (n.getElementsByTagName && n.getElementsByTagName("*").length > MAX_SUBTREE) break;
    const reads = readRegion(n);
    levels.push({ el: n, all: reads.length, withPick: reads.filter(r => r.pick).length });
    n = n.parentElement;
  }
  if (!levels.length) return el;

  // Deepest-first, so the first hit is always the smallest qualifying element.
  const bestPick = Math.max.apply(null, levels.map(l => l.withPick));
  if (bestPick >= 2) {
    for (const l of levels) if (l.withPick === bestPick) return l.el;
  }
  // Content cannot discriminate when the log is nearly empty -- and AT THE START
  // OF A DRAFT it is empty, which is exactly when you bind it. Worse, an empty
  // list has no height, so you cannot click it at all; you click the panel
  // around it, or its heading. Climbing by match count from there finds nothing
  // and walks into a wrapper that also holds the available list.
  //
  // So fall back to structure. Working outward from the click, take the first
  // level that either IS a list or CONTAINS one, preferring the emptiest
  // candidate: at the start of a draft the pick log holds nothing while the
  // available list holds hundreds, and anything over MAX_REGION is excluded
  // outright, so the two separate cleanly with no content to go on.
  if (bestPick < 2) {
    const SEL = "tbody,ul,ol,[role=rowgroup],[role=list]";
    const listy = e => /^(TBODY|UL|OL|TABLE|DL)$/.test(e.tagName) ||
      ["rowgroup", "list", "listbox", "table", "grid"]
        .indexOf((e.getAttribute("role") || "").toLowerCase()) >= 0;
    for (const l of levels) {
      if (listy(l.el) && l.all <= MAX_REGION) return l.el;
      if (!l.el.querySelectorAll) continue;
      const inner = Array.prototype.slice.call(l.el.querySelectorAll(SEL))
        .map(x => ({ el: x, n: readRegion(x).length }))
        .filter(x => x.n <= MAX_REGION);
      if (inner.length) { inner.sort((a, b) => a.n - b.n); return inner[0].el; }
    }
  }
  // Otherwise settle for the smallest ancestor that sees more than one player.
  for (const l of levels) if (l.all >= 2 && l.all <= MAX_REGION) return l.el;
  let best = levels[0];
  for (const l of levels) if (l.all > best.all && l.all <= MAX_REGION) best = l;
  return best.el;
}

// ---- finding the list without being told where it is ----
// Aiming is the weak link: you click a name or a bye-week line, and everything
// depends on climbing out of it correctly. This starts from the opposite end
// and needs no click at all.
//
// Every element whose OWN text is just a small integer is a candidate pick
// number. The pick log is then the smallest ancestor of those that sees the
// most numbered player rows. The available-players table cannot win: its rank
// column is a bare integer too, but its rows put the NAME first, so not one of
// them parses a pick number, and the table as a whole reads far more players
// than MAX_REGION.
function autoFindPickList() {
  const seeds = [];
  const all = document.body.getElementsByTagName("*");
  for (let i = 0; i < all.length && seeds.length < 300; i++) {
    const e = all[i];
    if (e.children.length) continue;                 // leaves only, so this stays cheap
    if (HOST && HOST.contains(e)) continue;          // never our own panel
    const t = (e.textContent || "").trim();
    if (!/^\d{1,3}$/.test(t)) continue;
    const v = +t;
    if (v >= 1 && v <= 150) seeds.push(e);
  }

  const tried = new Set();
  let numbered = null, plain = null;
  const consider = n => {
    if (tried.has(n)) return true;
    tried.add(n);
    if (n.getElementsByTagName("*").length > MAX_SUBTREE) return false;  // stop climbing
    const reads = readRegion(n);
    if (reads.length < 2 || reads.length > MAX_REGION) return true;
    const withPick = reads.filter(r => r.pick).length;
    const cand = { el: n, withPick: withPick, all: reads.length };
    if (withPick >= 2) {
      if (!numbered || cand.withPick > numbered.withPick ||
          (cand.withPick === numbered.withPick && cand.all < numbered.all)) numbered = cand;
    } else if (!plain || cand.all < plain.all) {
      plain = cand;
    }
    return true;
  };

  for (const seed of seeds) {
    let n = seed.parentElement;
    for (let up = 0; up < MAX_CLIMB && n && n !== document.body; up++, n = n.parentElement) {
      if (!consider(n)) break;
    }
  }
  // A numbered list is a pick log. A list with no numbers might be one too --
  // some rooms only number the current pick -- so it is offered second rather
  // than not at all.
  const best = numbered || plain;
  return best ? { el: best.el, picks: best.withPick, players: best.all } : null;
}

function bindTo(el, how) {
  BR.el = el;
  BR.path = elPath(el);
  // Everything already visible here is the state BEFORE we started watching.
  // Seed it as "seen" but do not commit it; joining mid-draft, the user imports
  // it explicitly with one tap.
  const found = scanRegion();
  BR.pending = [];
  BR.preload = found;
  BR.note = describeBind(found.length, how);
  renderBridgePanel();
  renderBridge();
  return found.length;
}

// The panel used to say "ARM and the next pick will come in automatically" even
// when it was already armed and there was no ARM button to press -- which is
// exactly the sort of thing you cannot debug at 9:31pm.
function describeBind(n, how) {
  const where = how ? how + " " : "";
  if (!n) {
    return where + "Bound, but it reads nothing there yet. " +
      (BR.armed ? "Already live — a pick appearing in it will come in by itself. "
                : "ARM and the next pick will come in automatically. ") +
      "If picks are already showing on the board, this is the wrong region: " +
      "tap FIND IT, or DIAGNOSE to see what it is looking at.";
  }
  return where + "Bound. " + n + " player" + (n > 1 ? "s" : "") + " already on the board — " +
    "IMPORT " + (n > 1 ? "them" : "him") +
    (BR.armed ? "." : ", or ARM to only take picks from here on.");
}

// One pass over the bound container. Returns the reads it has not seen before.
function scanRegion() {
  if (!BR.el || !document.contains(BR.el)) {
    const again = elFromPath(BR.path);
    if (!again) { BR.note = "lost the board region — rebind it"; return []; }
    BR.el = again;
  }
  const all = readRegion(BR.el);
  let maxPick = 0;
  for (const r of all) if (r.pick && r.pick > maxPick && r.pick <= 150) maxPick = r.pick;
  if (maxPick) setBoardPick(maxPick + 1);
  return all.filter(r => !BR.seen.has(r.player.n));
}

// ---- committing ----
function bridgeTick() {
  if (!BR.armed || BR.paused) return;
  const fresh = scanRegion();
  if (!fresh.length) {
    // The status line at the top only redraws when something calls it, so
    // without this the panel sat on "Board bridge not connected" while the
    // bridge was in fact live and simply had nothing new to report.
    if (BR.el) { const first = !bridgeSeen; bridgeSeen = Date.now(); if (first) renderBridge(); }
    return;
  }
  bridgeSeen = Date.now();

  // A burst means the container is probably the available-players list, not the
  // pick log. Committing 200 players would destroy the board, so stop instead.
  if (fresh.length > BURST_LIMIT && BR.lastCount > 0) {
    BR.paused = true;
    BR.note = fresh.length + " new names at once — that looks like the available " +
              "list, not the pick log. Nothing was committed.";
    renderBridgePanel();
    return;
  }

  const myPickSet = new Set(myPicks());
  for (const f of fresh) {
    BR.seen.add(f.player.n);
    // Never auto-commit a shaky read. It goes to the confirm strip, where it is
    // one tap — still no typing, but a wrong guess cannot slip through.
    if (f.conf < AUTO_COMMIT) { BR.pending.push(f); continue; }
    const isMine = f.pick ? myPickSet.has(f.pick) : false;
    ingestPick(f.player.n, { src: "board", conf: f.conf, pick: f.pick || undefined,
                             mine: isMine, note: f.stale });
  }
  BR.lastCount = BR.seen.size;
  saveBridge();
  render();
  renderBridgePanel();
}

function confirmPending(i, mineFlag) {
  const f = BR.pending[i];
  if (!f) return;
  BR.pending.splice(i, 1);
  ingestPick(f.player.n, { src: "confirmed", conf: 1, pick: f.pick || undefined, mine: !!mineFlag });
  saveBridge();
  render();
  renderBridgePanel();
}
function rejectPending(i) {
  const f = BR.pending[i];
  if (!f) return;
  BR.pending.splice(i, 1);
  BR.seen.delete(f.player.n);   // let it be re-read if it really was a pick
  renderBridgePanel();
}

function saveBridge() {
  try {
    localStorage.setItem(BR_KEY, JSON.stringify({ path: BR.path, seen: [...BR.seen], armed: BR.armed }));
  } catch (e) {}
}
function loadBridge() {
  try {
    const d = JSON.parse(localStorage.getItem(BR_KEY) || "null");
    if (!d) return;
    BR.path = d.path || null;
    BR.seen = new Set(d.seen || []);
    BR.armed = !!d.armed;
    BR.lastCount = BR.seen.size;
    if (BR.path) BR.el = elFromPath(BR.path);
  } catch (e) {}
}

// ---- binding the region by clicking it ----
let picking = false, hiBox = null;
let reopenAfterPick = false;
function startPick() {
  if (picking) return;
  picking = true;
  BR.note = "Tap the part of the page that lists the picks already made.";
  renderBridgePanel();
  // On a phone the panel is a full-screen overlay, so it is covering the very
  // thing you are being asked to tap. Get out of the way, then come back.
  reopenAfterPick = false;
  if (typeof PANEL !== "undefined" && PANEL.narrow() && PANEL.isOpen()) {
    reopenAfterPick = true;
    PANEL.set(false);
  }
  hiBox = document.createElement("div");
  hiBox.style.cssText = "position:fixed;z-index:2147483646;pointer-events:none;" +
    "border:2px solid #F5C518;background:rgba(245,197,24,.14);border-radius:4px";
  document.body.appendChild(hiBox);
  // pointermove covers mouse, pen and touch-drag; mousemove alone gives a
  // phone no highlight at all.
  document.addEventListener("pointermove", onHover, true);
  document.addEventListener("mousemove", onHover, true);
  document.addEventListener("click", onPickClick, true);
  document.addEventListener("keydown", onPickKey, true);
}
function stopPick() {
  picking = false;
  if (hiBox) { hiBox.remove(); hiBox = null; }
  document.removeEventListener("pointermove", onHover, true);
  document.removeEventListener("mousemove", onHover, true);
  document.removeEventListener("click", onPickClick, true);
  document.removeEventListener("keydown", onPickKey, true);
  if (reopenAfterPick && typeof PANEL !== "undefined") { reopenAfterPick = false; PANEL.set(true); }
}
let hoverLast = 0, hoverEl = null;
function onHover(e) {
  const el = e.target;
  if (!el || el === hiBox || HOST.contains(el)) return;
  // Outline what would ACTUALLY be bound, not the leaf under the cursor, so
  // there is no surprise about what got picked. Resolving now climbs up to 16
  // levels and reads each one, which is far too much work to redo on every
  // pixel of pointer movement.
  const now = Date.now();
  if (el === hoverEl && now - hoverLast < 120) return;
  hoverEl = el; hoverLast = now;
  const r = resolveContainer(el).getBoundingClientRect();
  hiBox.style.left = r.left + "px"; hiBox.style.top = r.top + "px";
  hiBox.style.width = r.width + "px"; hiBox.style.height = r.height + "px";
}
function onPickKey(e) { if (e.key === "Escape") { stopPick(); BR.note = ""; renderBridgePanel(); } }
function onPickClick(e) {
  if (HOST.contains(e.target)) return;
  e.preventDefault(); e.stopPropagation();
  const el = resolveContainer(e.target);
  stopPick();
  if (bindTo(el, "") === 0) {
    // Climbing found nothing where you clicked. Rather than sit there saying
    // "bound" and reading zero, go and look for the list.
    const auto = autoFindPickList();
    if (auto && auto.el !== el) bindTo(auto.el, "Your tap read nothing, so I found the list myself:");
  }
}

function findItForMe() {
  BR.note = "Looking for the pick list…";
  renderBridgePanel();
  const auto = autoFindPickList();
  if (!auto) {
    BR.note = "Could not find a pick list on this page. If picks are showing, " +
      "tap BIND THE PICK LIST and click directly on one of them, then DIAGNOSE.";
    renderBridgePanel();
    return;
  }
  bindTo(auto.el, "Found a list of " + auto.players + " player" + (auto.players > 1 ? "s" : "") +
    (auto.picks ? " with pick numbers." : " (no pick numbers in it).") +
    " Check it with DIAGNOSE if anything looks off.");
}
function importPreload() {
  const list = BR.preload || [];
  const myPickSet = new Set(myPicks());
  // Import in board order so the pick numbers line up.
  list.sort((a, b) => (a.pick || 999) - (b.pick || 999));
  for (const f of list) {
    BR.seen.add(f.player.n);
    const isMine = f.pick ? myPickSet.has(f.pick) : false;
    ingestPick(f.player.n, { src: "import", conf: f.conf, pick: f.pick || undefined, mine: isMine });
  }
  BR.preload = [];
  BR.armed = true;
  BR.lastCount = BR.seen.size;
  BR.note = "Imported " + list.length + " picks. Watching for new ones.";
  saveBridge(); render(); renderBridgePanel();
}
function armOnly() {
  (BR.preload || []).forEach(f => BR.seen.add(f.player.n));
  BR.preload = [];
  BR.armed = true;
  BR.paused = false;
  BR.lastCount = BR.seen.size;
  BR.note = "Watching. New picks will appear automatically.";
  saveBridge(); renderBridgePanel();
}

// ---- the bridge's own panel, inside the shadow root ----
// ---- what am I actually reading? ----
// The failure that costs a draft is a bridge that says LIVE and commits
// nothing: there is no error to read, and no way to tell whether the region is
// wrong, the rows are wrong, or the names are wrong. This prints the reader's
// own view of the bound region -- the raw row text, what survived stripping,
// and what it matched -- on the panel, because nobody opens a console at pick
// three. It reads only; it commits nothing.
function brEsc(s) {
  return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
}
function brTag(el) {
  const cls = String(el.className || "").split(/\s+/)[0];
  return el.tagName.toLowerCase() + (el.id ? "#" + el.id : cls ? "." + cls : "");
}
let brDiag = null;
function diagnose() {
  if (brDiag) { brDiag = null; renderBridgePanel(); return; }
  if (!BR.el) {
    BR.note = "Nothing is bound yet — tap BIND THE PICK LIST first.";
    renderBridgePanel();
    return;
  }
  const rows = rowsOf(BR.el);
  const lines = [];
  for (const row of rows) {
    if (lines.length >= 8) break;
    const info = parseRow(row.text);
    let verdict;
    if (!plausibleName(info.name)) {
      verdict = "no name left after stripping";
    } else {
      const m = matchBoardName(info.name, { pos: info.pos, team: info.team });
      const pc = m.p ? Math.round(m.conf * 100) + "%" : "";
      verdict = !m.p ? "no match for \u201C" + info.name + "\u201D"
        : m.conf < MIN_READ ? "too weak — " + m.p.n + " " + pc
        : m.p.n + " " + pc + (info.pick ? "  pick " + info.pick : "  NO PICK NUMBER");
    }
    lines.push({ raw: row.text.slice(0, 70), verdict: verdict });
  }
  brDiag = {
    tag: brTag(BR.el),
    rows: rows.length,
    reads: readRegion(BR.el).length,
    lines: lines,
  };
  renderBridgePanel();
}

function renderBridgePanel() {
  const box = UIROOT.getElementById("brPanel");
  if (!box) return;
  const state = !BR.el ? "unbound" : BR.paused ? "paused" : BR.armed ? "live" : "bound";
  let h = '<div class="brhd">BOARD BRIDGE — <b>' + state.toUpperCase() + '</b></div>';
  if (BR.note) h += '<div class="brnote">' + BR.note + '</div>';
  h += '<div class="brbtns">';
  h += '<button id="brBind">' + (BR.el ? "REBIND REGION" : "BIND THE PICK LIST") + '</button>';
  if (BR.preload && BR.preload.length) h += '<button id="brImport">IMPORT ' + BR.preload.length + '</button>';
  if (BR.el && !BR.armed) h += '<button id="brArm">ARM</button>';
  if (BR.armed && !BR.paused) h += '<button id="brPause">PAUSE</button>';
  if (BR.paused) h += '<button id="brResume">RESUME</button>';
  h += '<button id="brFind">FIND IT</button>';
  h += '<button id="brRescan">RESCAN</button>';
  h += '<button id="brWhy">' + (brDiag ? "HIDE DIAG" : "DIAGNOSE") + '</button>';
  h += '</div>';
  // Always visible, so "LIVE but reading nothing" can never look like "working".
  if (BR.el) {
    const live = readRegion(BR.el).length;
    h += '<div class="brpr" style="margin-top:6px">watching &lt;' + brEsc(brTag(BR.el)) +
      '&gt; — <b>' + live + '</b> player' + (live === 1 ? '' : 's') + ' readable in it' +
      (live ? '' : '  ← that is the problem') + '</div>';
  }
  if (brDiag) {
    h += '<div class="brhd" style="margin-top:8px">WHAT THE READER SEES</div>';
    h += '<div class="brpr">bound &lt;' + brEsc(brDiag.tag) + '&gt; — ' + brDiag.rows +
         ' rows, ' + brDiag.reads + ' recognised</div>';
    brDiag.lines.forEach(l => {
      h += '<div class="brp" style="padding-top:6px;margin-top:6px">' +
        '<div class="brpr" style="margin-bottom:2px">' + brEsc(l.raw) + '</div>' +
        '<div class="brpn" style="font-size:11px">&rarr; ' + brEsc(l.verdict) + '</div></div>';
    });
  }
  if (BR.pending.length) {
    h += '<div class="brhd" style="margin-top:8px">NEEDS ONE TAP — not sure enough to commit</div>';
    BR.pending.forEach((f, i) => {
      h += '<div class="brp"><div class="brpn">' + f.player.n +
        ' <span class="pos ' + f.player.p + '">' + f.player.p + '</span>' +
        ' <span class="brpc">' + Math.round(f.conf * 100) + '%</span></div>' +
        '<div class="brpr">read: “' + f.raw.slice(0, 60) + '”</div>' +
        '<div class="brbtns"><button data-ok="' + i + '">GONE</button>' +
        '<button data-mine="' + i + '">MINE</button>' +
        '<button data-no="' + i + '">NOT HIM</button></div></div>';
    });
  }
  box.innerHTML = h;
  const B = id => UIROOT.getElementById(id);
  if (B("brBind")) B("brBind").onclick = startPick;
  if (B("brImport")) B("brImport").onclick = importPreload;
  if (B("brArm")) B("brArm").onclick = armOnly;
  if (B("brPause")) B("brPause").onclick = () => { BR.paused = true; renderBridgePanel(); };
  if (B("brResume")) B("brResume").onclick = () => { BR.paused = false; BR.note = ""; renderBridgePanel(); };
  if (B("brFind")) B("brFind").onclick = findItForMe;
  if (B("brRescan")) B("brRescan").onclick = () => { BR.lastCount = 0; bridgeTick(); };
  if (B("brWhy")) B("brWhy").onclick = diagnose;
  box.querySelectorAll("[data-ok]").forEach(b => b.onclick = () => confirmPending(+b.getAttribute("data-ok"), false));
  box.querySelectorAll("[data-mine]").forEach(b => b.onclick = () => confirmPending(+b.getAttribute("data-mine"), true));
  box.querySelectorAll("[data-no]").forEach(b => b.onclick = () => rejectPending(+b.getAttribute("data-no")));
}

// ---- run ----
// A MutationObserver catches the common case instantly. The interval is the
// backstop for the case that actually bites: Yahoo replacing the whole subtree,
// which detaches the observer without any error.
loadBridge();

// Try to find the pick list without being asked. Nothing is committed by this
// -- binding only decides where to LOOK; importing and arming are still
// deliberate taps -- so the worst case is a wrong region you rebind, and the
// best case is that draft night needs no setup at all. Retried a few times
// because the draft room renders its sidebar well after the script runs.
[1500, 5000, 12000].forEach(ms => setTimeout(() => {
  if (BR.el || picking) return;
  const auto = autoFindPickList();
  if (!auto) return;
  bindTo(auto.el, "Found the pick list on my own (" + auto.players + " player" +
    (auto.players > 1 ? "s" : "") + " in it).");
}, ms));

new MutationObserver(() => { if (BR.armed && !BR.paused) bridgeTick(); })
  .observe(document.body, { childList: true, subtree: true, characterData: true });
setInterval(() => { if (BR.armed && !BR.paused) bridgeTick(); }, 1500);
renderBridgePanel();
if (BR.armed) bridgeTick();

// Deliberate escape hatch. The browser tests drive the real UI, but they assert
// against this; and on draft night it is the fastest way to see what the reader
// is actually reading when something looks wrong. Console-only, no UI.
window.__sstlv = {
  BR, scanRegion, bridgeTick, matchBoardName, searchPlayers, ingestPick, diagnose,
  autoFindPickList, findItForMe, resolveContainer,
  curPick, desync, likelyNext,
  state: () => ({
    bound: !!BR.el, armed: BR.armed, paused: BR.paused,
    pick: curPick(), gone: gone.size, mine: mine.slice(),
    seen: [...BR.seen], pending: BR.pending.map(p => p.player.n),
    feed: feed.slice(-10), note: BR.note,
  }),
};
